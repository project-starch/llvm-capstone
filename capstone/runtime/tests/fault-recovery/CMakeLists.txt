cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_LIST_DIR}/../../../ports/common/cmake/Port.cmake")
project(capstone_fault_recovery LANGUAGES C)
port_check_platform()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
find_package(Python3 3.11.4 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${CAPSTONE_REPO_ROOT}/capstone/runtime" "${CMAKE_BINARY_DIR}/capstone-runtime")

if(PORT_PLATFORM STREQUAL "capstone-domain")
  enable_language(ASM)
  if(NOT CAPSTONE_DOMAIN_FAULT_RECOVERY)
    message(FATAL_ERROR "This suite requires CAPSTONE_DOMAIN_FAULT_RECOVERY=ON")
  endif()
  foreach(kind RANGE 0 6)
    add_executable(fault-probe-${kind} domain.c)
    target_compile_definitions(fault-probe-${kind} PRIVATE RUNTIME_FAULT_KIND=${kind})
    target_compile_options(fault-probe-${kind} PRIVATE -ffunction-sections -fdata-sections)
    capstone_configure_domain(fault-probe-${kind} DATA_BYTES 16384 STACK_BYTES 16384)
  endforeach()
  set(CAPSTONE_FAULT_LINUX_BUILD "" CACHE PATH "Matching Linux guest test build")
  if(BUILD_TESTING)
    add_test(NAME qemu-runtime-fault-recovery COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/run.py" --domain-build "${CMAKE_BINARY_DIR}"
      --linux-build "${CAPSTONE_FAULT_LINUX_BUILD}" --output "${CMAKE_BINARY_DIR}/results")
    set_tests_properties(qemu-runtime-fault-recovery PROPERTIES RUN_SERIAL TRUE TIMEOUT 600
      ENVIRONMENT "CAPSTONE_REPO_ROOT=${CAPSTONE_REPO_ROOT}")
  endif()
elseif(PORT_PLATFORM STREQUAL "linux-guest")
  set(libcapstone "${CAPSTONE_BUILDROOT_DIR}/package/modcapstone/userspace/lib")
  foreach(program reentry launcher)
    add_executable(fault-${program} ${program}.c "${libcapstone}/libcapstone.c")
    target_include_directories(fault-${program} PRIVATE "${libcapstone}"
      "${CAPSTONE_BUILDROOT_DIR}/package/modcapstone/include")
    target_link_libraries(fault-${program} PRIVATE Capstone::LinuxFaultPolicy)
  endforeach()
  add_executable(fault-supervisor supervisor.c)
else()
  add_executable(fault-policy-test policy-test.c)
  target_link_libraries(fault-policy-test PRIVATE Capstone::LinuxFaultPolicy)
  add_test(NAME linux-fault-policy COMMAND fault-policy-test)
  add_test(NAME fault-verdict-controls COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/check-verdicts.py")
  add_test(NAME domain-build-controls COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/check-build.py")
  set_tests_properties(linux-fault-policy fault-verdict-controls domain-build-controls
    PROPERTIES TIMEOUT 60 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
