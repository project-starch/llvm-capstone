# The reuse study: PostgreSQL 17.5 single-user backend on work.sql against the 16-byte MAXALIGN cluster; it ends at
# end of input, so no signal is needed. Both arms in one boot. The backend finds its share directory (timezone sets)
# beside its bin directory, so the images sit in bin/ and share/ holds the native install's data files.
python3 capstone/runtime/virtual/trap_platform.py run --kit "$KIT" --work "$OUT/reuse/postgres" --step-timeout 3600 --file pgcluster="$PREBUILT/reuse/pg-fixture/cluster" --file work.sql=capstone/ports/postgres/app/work.sql --file share="$PREBUILT/reuse/pg-share" --file bin/postgres-stock.dom="$PREBUILT/reuse/postgres-stock/image/postgres.dom" --file bin/postgres-protected.dom="$PREBUILT/reuse/postgres-protected/image/postgres.dom" --step "pg-stock=rm -rf /tmp/pgcluster; cp -a /mnt/vm/pgcluster /tmp/pgcluster && chown -R nobody /tmp/pgcluster && su nobody -s /bin/sh -c 'cd /mnt/vm; ./capstone-vexec ./bin/postgres-stock.dom --single -D /tmp/pgcluster -c shared_buffers=4MB -c max_connections=10 -c timezone=GMT -c log_timezone=GMT -c dynamic_shared_memory_type=sysv -c exit_on_error=true postgres < work.sql'" --step "pg-protected=rm -rf /tmp/pgcluster; cp -a /mnt/vm/pgcluster /tmp/pgcluster && chown -R nobody /tmp/pgcluster && su nobody -s /bin/sh -c 'cd /mnt/vm; ./capstone-vexec ./bin/postgres-protected.dom --single -D /tmp/pgcluster -c shared_buffers=4MB -c max_connections=10 -c timezone=GMT -c log_timezone=GMT -c dynamic_shared_memory_type=sysv -c exit_on_error=true postgres < work.sql'"
