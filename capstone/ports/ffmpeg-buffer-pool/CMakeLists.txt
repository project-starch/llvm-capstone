cmake_minimum_required(VERSION 3.25)

# One compiler per build tree. The Linux loader and Capstone domains have
# different ABIs, even though both execute inside the same QEMU guest.
set(FFPOOL_PLATFORM native CACHE STRING "native, capstone-domain or linux-host")
set_property(CACHE FFPOOL_PLATFORM PROPERTY STRINGS native capstone-domain linux-host)
get_filename_component(CAPSTONE_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." REALPATH)
file(REAL_PATH "${CMAKE_BINARY_DIR}" build_path)
cmake_path(IS_PREFIX CAPSTONE_REPO_ROOT "${build_path}" NORMALIZE in_repository)
if(in_repository)
  message(FATAL_ERROR "Keep generated builds and downloaded FFmpeg sources outside the repository; use a preset or -B /tmp/ffpool-build.")
endif()
project(ffmpeg_buffer_pool LANGUAGES C)

if(NOT FFPOOL_PLATFORM MATCHES "^(native|capstone-domain|linux-host)$")
  message(FATAL_ERROR "Unknown FFPOOL_PLATFORM: ${FFPOOL_PLATFORM}")
endif()
if(NOT FFPOOL_PLATFORM STREQUAL "native" AND NOT CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Select the ${FFPOOL_PLATFORM} toolchain with its configure preset.")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

include(cmake/Workspace.cmake)
include(CTest)

if(FFPOOL_PLATFORM STREQUAL "linux-host")
  add_subdirectory(capstone/linux-host)
else()
  find_package(Python3 3.11.4 REQUIRED COMPONENTS Interpreter)
  include(cmake/FFmpegSource.cmake)
  include(cmake/Replay.cmake)
  if(FFPOOL_PLATFORM STREQUAL "native")
    option(FFPOOL_BUILD_DECODERS "Build stock and instrumented native FFmpeg" ON)
    if(FFPOOL_BUILD_DECODERS)
      add_subdirectory(native/ffmpeg)
    endif()
  endif()
  if(BUILD_TESTING)
    include(cmake/Tests.cmake)
  endif()
endif()
