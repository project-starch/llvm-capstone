cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_LIST_DIR}/../../common/cmake/Port.cmake")
project(apr_pools LANGUAGES C)
port_check_platform()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
find_package(Python3 3.11.4 REQUIRED COMPONENTS Interpreter)
port_path(APRP_WORK "${tmp_root}/apr-pools" "Sources, builds and results")
if(PORT_PLATFORM STREQUAL "linux-guest")
  add_subdirectory(src/linux-guest)
else()
  include(cmake/APRSource.cmake)
  include(cmake/Pools.cmake)
  if(BUILD_TESTING AND PORT_PLATFORM STREQUAL "native")
    port_add_support_tests()
    add_test(NAME source-preparation COMMAND "${Python3_EXECUTABLE}"
      "${PROJECT_SOURCE_DIR}/tests/native/test-source.py" "${APRP_ARCHIVE}" "${APRP_SOURCE}")
    add_test(NAME allocator-example COMMAND allocator-example)
    if(APRP_BUCKETS)
      add_test(NAME apr-util-source-preparation COMMAND "${Python3_EXECUTABLE}"
        "${PROJECT_SOURCE_DIR}/tests/native/test-apr-util-source.py" "${APU_ARCHIVE}" "${APU_SOURCE}")
      add_test(NAME bucket-example COMMAND bucket-example)
    endif()
  endif()
endif()
