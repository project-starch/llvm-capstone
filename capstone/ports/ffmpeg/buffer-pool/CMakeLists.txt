cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_LIST_DIR}/../../common/cmake/Port.cmake")
project(ffmpeg_buffer_pool LANGUAGES C)

port_check_platform()
if(PORT_PLATFORM STREQUAL "cheribsd")
  set(FFPOOL_CHERI ON)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

include(cmake/Workspace.cmake)
include(CTest)

if(PORT_PLATFORM STREQUAL "linux-guest")
  add_subdirectory(src/linux-guest)
else()
  find_package(Python3 3.11.4 REQUIRED COMPONENTS Interpreter)
  include(cmake/FFmpegSource.cmake)
  include(cmake/Replay.cmake)
  if(PORT_PLATFORM STREQUAL "native")
    option(FFPOOL_BUILD_DECODERS "Build stock and instrumented native FFmpeg" ON)
    if(FFPOOL_BUILD_DECODERS)
      add_subdirectory(src/native/ffmpeg)
    endif()
  endif()
  if(BUILD_TESTING)
    include(cmake/Tests.cmake)
  endif()
endif()
