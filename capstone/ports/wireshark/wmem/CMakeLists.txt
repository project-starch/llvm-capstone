cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_LIST_DIR}/../../common/cmake/Port.cmake")
project(wireshark_wmem LANGUAGES C)
port_check_platform()
if(NOT PORT_HOSTED)
  message(FATAL_ERROR "the wmem port builds hosted programs only: capstone-application, cheribsd or native")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
find_package(Python3 3.11.4 REQUIRED COMPONENTS Interpreter)
port_path(WM_WORK "${tmp_root}/wireshark-wmem" "External sources, builds and results")
# The nested arm in a Capstone process: patch 0001 makes every object of the
# block and block_fast allocators a child lifetime of its block (CDERIVE),
# revoked by free, realloc and the reset that ends its pool (CREVOKE).
option(WM_SUBLET "wmem objects as Sublet lifetimes (patch 0001)" OFF)
if(WM_SUBLET AND NOT PORT_PLATFORM STREQUAL "capstone-application")
  message(FATAL_ERROR "WM_SUBLET needs the capstone-application toolchain")
endif()
include(cmake/Source.cmake)
include(cmake/Replay.cmake)
if(PORT_PLATFORM STREQUAL "native" AND BUILD_TESTING)
  port_add_support_tests()
  add_test(NAME replay COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/native/test-replay.py"
    "$<TARGET_FILE:replay>" "${CMAKE_BINARY_DIR}/tests")
  add_test(NAME source-preparation COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/native/test-source.py"
    "${WM_ARCHIVE}" "${CMAKE_BINARY_DIR}/source-sublet")
  add_test(NAME allocator-example COMMAND allocator-example)
endif()
